export type UploadStatus = 'idle' | 'uploading' | 'success' | 'error';

  name: string;
  size: number;
  sizeFormatted: string;
  type: string;
  extension: string;
  lastModified: Date;
  linesCount?: number;
  preview?: string;
  hash?: string;
}

export interface UploadedFile {
  id: string;
  file: File;
  status: UploadStatus;
  progress: number;
  analysis?: FileAnalysis;

}

import React, { useCallback, useRef, useState } from 'react';
import { UploadedFile } from './types';
import { analyzeFile } from './analyzer';

interface Props {
  onFilesAdded: (files: UploadedFile[]) => void;
}

export const FileUploader: React.FC<Props> = ({ onFilesAdded }) => {
  const [isDragging, setIsDragging] = useState(false);
  const inputRef = useRef<HTMLInputElement>(null);

  const processFiles = useCallback(
    async (fileList: FileList) => {
      const items: UploadedFile[] = Array.from(fileList).map((file) => ({
        id: crypto.randomUUID(),
        file,
        status: 'uploading',
        progress: 0,
      }));

      onFilesAdded(items);

      
      for (const item of items) {
        for (let p = 0; p <= 100; p += 20) {
          await new Promise((r) => setTimeout(r, 100));
          item.progress = p;
        }
        try {
          item.analysis = await analyzeFile(item.file);
          item.status = 'success';
        } catch (e) {
          item.status = 'error';
          item.error = e instanceof Error ? e.message : 'Ошибка анализа';
        }
      }
      onFilesAdded([...items]); 
    },
    [onFilesAdded],
  );

  const handleDrop = (e: React.DragEvent) => {
    e.preventDefault();
    setIsDragging(false);
    if (e.dataTransfer.files.length) processFiles(e.dataTransfer.files);
  };

  return (
    <div
      onDragOver={(e) => {
        e.preventDefault();
        setIsDragging(true);
      }}
      onDragLeave={() => setIsDragging(false)}
      onDrop={handleDrop}
      onClick={() => inputRef.current?.click()}
      style={{
        border: `2px dashed ${isDragging ? '#2563eb' : '#94a3b8'}`,
        borderRadius: 12,
        padding: 40,
        textAlign: 'center',
        cursor: 'pointer',
        background: isDragging ? '#eff6ff' : '#f8fafc',
        transition: 'all 0.2s',
      }}
    >
      <input
        ref={inputRef}
        type="file"
        multiple
        hidden
        onChange={(e) => e.target.files && processFiles(e.target.files)}
      />
      <p style={{ margin: 0, fontSize: 16, color: '#334155' }}>
        Перетащите файлы сюда или <strong>нажмите для выбора</strong>
      </p>
    </div>
  );
};

import React, { useState } from 'react';
import { UploadedFile } from './types';
import { FileUploader } from './FileUploader';
import { FileResults } from './FileResults';

export const App: React.FC = () => {
  const [files, setFiles] = useState<UploadedFile[]>([]);

  const handleAdd = (newItems: UploadedFile[]) => {
    setFiles((prev) => {
      const map = new Map(prev.map((f) => [f.id, f]));
      newItems.forEach((f) => map.set(f.id, f));
      return Array.from(map.values());
    });
  };

  const handleRemove = (id: string) =>
    setFiles((prev) => prev.filter((f) => f.id !== id));

  return (
    <div style={{ maxWidth: 720, margin: '40px auto', fontFamily: 'sans-serif' }}>
      <h1>Загрузка и анализ файлов</h1>
      <FileUploader onFilesAdded={handleAdd} />
      <FileResults files={files} onRemove={handleRemove} />
    </div>
  );
};


import ast
import os
from typing import Dict, Any, List

class PythonCodeAnalyzer:
    def __init__(self, max_line_length: int = 79):
        """Инициализация анализатора."""
        self.max_line_length = max_line_length

    def analyze_file(self, file_path: str) -> Dict[str, Any]:
        stats = {
            'file': file_path,
            'total_lines': 0,
            'empty_lines': 0,
            'comment_lines': 0,
            'functions': 0,
            'classes': 0,
            'long_lines': 0,
            'errors': []
        }

        # 1. Чтение файла и обработка ошибок чтения
        try:
            with open(file_path, 'r', encoding='utf-8') as file:
                content = file.read()
                lines = content.splitlines()
        except FileNotFoundError:
            stats['errors'].append("Файл не найден.")
            return stats
        except UnicodeDecodeError:
            stats['errors'].append("Ошибка кодировки (файл не в UTF-8).")
            return stats
        except PermissionError:
            stats['errors'].append("Нет прав для чтения файла.")
            return stats
        except Exception as e:
            stats['errors'].append(f"Непредвиденная ошибка: {e}")
            return stats

        # 2. Подсчет строк, пустых строк и комментариев
        stats['total_lines'] = len(lines)
        for line in lines:
            stripped_line = line.strip()
            if not stripped_line:
                stats['empty_lines'] += 1
            elif stripped_line.startswith('#'):
                stats['comment_lines'] += 1

        # 4. Проверка длины строк
        for line in lines:
            if len(line) > self.max_line_length:
                stats['long_lines'] += 1

        # 3. Подсчет функций и классов через AST
        try:
            tree = ast.parse(content)
            for node in ast.walk(tree):
                if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
                    stats['functions'] += 1
                elif isinstance(node, ast.ClassDef):
                    stats['classes'] += 1
        except SyntaxError as e:
            stats['errors'].append(f"Синтаксическая ошибка Python: {e}")

        return stats

    def analyze_directory(self, dir_path: str) -> List[Dict[str, Any]]:
        results = []
        if not os.path.isdir(dir_path):
            print(f"Ошибка: Директория '{dir_path}' не найдена.")
            return results

        for root, _, files in os.walk(dir_path):
            for file in files:
                if file.endswith('.py'):
                    full_path = os.path.join(root, file)
                    results.append(self.analyze_file(full_path))
        return results


def print_report(stats_list: List[Dict[str, Any]]):
    """Выводит результаты анализа в консоль."""
    print("-" * 60)
    print(f"{'ФАЙЛ':<30} | {'Строки':<8} | {'Ф/К':<6} | {'Ошибки'}")
    print("-" * 60)
    
    total_lines = 0
    total_funcs = 0
    total_classes = 0
    
    for stats in stats_list:
        if stats['errors']:
            err_msg = "; ".join(stats['errors'])
            print(f"{os.path.basename(stats['file']):<30} | {'ОШИБКА':<8} | {'ОШИБКА':<6} | {err_msg}")
        else:
            total_lines += stats['total_lines']
            total_funcs += stats['functions']
            total_classes += stats['classes']
            
            print(f"{os.path.basename(stats['file']):<30} | {stats['total_lines']:<8} | "
                  f"{stats['functions']}/{stats['classes']:<4} | "
                  f"Пустых: {stats['empty_lines']}, Комм: {stats['comment_lines']}, "
                  f"Длинных(>{stats.get('max_line_length', 79)}): {stats['long_lines']}")
                  
    print("-" * 60)
    print(f"ИТОГО: Строк - {total_lines} | Функций - {total_funcs} | Классов - {total_classes}")
    print("-" * 60)
